▌ 初始化指令
# 1. 建立專案
npm create vite@latest xuan-men-chuan
# 選 Vanilla + JavaScript

# 2. 安裝依賴
cd xuan-men-chuan
npm install phaser
npm install firebase

# 3. 啟動開發
npm run dev

# 4. 部署
npm install -g firebase-tools
firebase login
firebase deploy
